#include <iostream>
using namespace std;

// Point 클래스
class Point {
private:
    int x, y;
public:
    Point(int x = 0, int y = 0) : x(x), y(y) {}
    void show() const {
        cout << "(" << x << ", " << y << ")";
    }
};

// Shape 클래스
class Shape {
public:
    void Draw() const {
        cout << "도형 그린다" << endl;
    }
};

// Line 클래스
class Line : public Shape {
private:
    Point start, end;
public:
    Line(int x1, int y1, int x2, int y2) : start(x1, y1), end(x2, y2) {}
    void Draw() const {
        cout << "직선 그린다: ";
        start.show();
        cout << " -> ";
        end.show();
        cout << endl;
    }
};

// Circle 클래스
class Circle : public Shape {
private:
    Point center, boundary;
public:
    Circle(int x1, int y1, int x2, int y2) : center(x1, y1), boundary(x2, y2) {}
    void Draw() const {
        cout << "원 그린다: ";
        center.show();
        cout << " -> ";
        boundary.show();
        cout << endl;
    }
};

int main() {
    Circle a(1, 1, 5, 5);
    Line b(5, 5, 9, 9);

    a.Draw();
    b.Draw();

    return 0;
}
